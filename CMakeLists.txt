cmake_minimum_required(VERSION 3.20)
project(httpclient LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Which HTTP stacks to build. The library always has the utilities (URI, Base64, SHA-1, SHA-2, FNV-1a, OAuth);
# each backend adds its client factory. Only the backends for the platform in use can build.
if(WIN32)
  set(_default_curl OFF)
  set(_default_winhttp ON)
elseif(ANDROID)
  set(_default_curl OFF)
  set(_default_winhttp OFF)
else()
  set(_default_curl ON)
  set(_default_winhttp OFF)
endif()
option(HTTPCLIENT_CURL "Build the libcurl backend (CurlClientFactory)" ${_default_curl})
option(HTTPCLIENT_WINHTTP "Build the WinHTTP backend (WinHTTPClientFactory)" ${_default_winhttp})
option(HTTPCLIENT_ANDROID "Build the Android HttpURLConnection backend (AndroidClientFactory, needs JNI)" ${ANDROID})
option(HTTPCLIENT_CFNETWORK "Build the CFNetwork backend (iOSCFClientFactory, Apple only)" OFF)
option(HTTPCLIENT_NSURL "Build the NSMutableURLRequest backend (iOSClientFactory, Apple only)" OFF)
if(PROJECT_IS_TOP_LEVEL)
  set(_default_tests ON)
else()
  set(_default_tests OFF)
endif()
option(HTTPCLIENT_BUILD_TESTS "Build the tests" ${_default_tests})
option(HTTPCLIENT_INSTALL "Generate the install rules" ${PROJECT_IS_TOP_LEVEL})

set(HTTPCLIENT_UTILITY_SOURCES
  src/Base64.cpp src/Base64Url.cpp src/FNV.cpp src/OAuth.cpp src/SHA1.cpp src/URI.cpp src/sha2.cpp)

# URI folds the case of host names with utf8proc: a system copy if there is one, else fetched and built here.
option(HTTPCLIENT_FETCH_UTF8PROC "Fetch and build utf8proc when the system has none" ON)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(UTF8PROC QUIET IMPORTED_TARGET libutf8proc)
endif()
if(TARGET PkgConfig::UTF8PROC)
  set(_utf8proc_target PkgConfig::UTF8PROC)
  set(_utf8proc_system TRUE)
elseif(HTTPCLIENT_FETCH_UTF8PROC)
  include(FetchContent)
  set(UTF8PROC_INSTALL OFF CACHE BOOL "" FORCE)
  set(UTF8PROC_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS_SAVED ${BUILD_SHARED_LIBS})
  set(BUILD_SHARED_LIBS OFF)
  FetchContent_Declare(utf8proc GIT_REPOSITORY https://github.com/JuliaStrings/utf8proc GIT_TAG v2.9.0)
  FetchContent_MakeAvailable(utf8proc)
  set(BUILD_SHARED_LIBS ${BUILD_SHARED_LIBS_SAVED})
  set(_utf8proc_target utf8proc)
  set(_utf8proc_system FALSE)
else()
  message(FATAL_ERROR "httpclient needs utf8proc (libutf8proc-dev): install it, or set HTTPCLIENT_FETCH_UTF8PROC=ON")
endif()

add_library(httpclient ${HTTPCLIENT_UTILITY_SOURCES})
target_link_libraries(httpclient PRIVATE ${_utf8proc_target})
add_library(httpclient::httpclient ALIAS httpclient)
target_compile_features(httpclient PUBLIC cxx_std_17)
# The headers include each other, and a host project's, with angle brackets: `#include <HTTPClient.h>`.
target_include_directories(httpclient PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include/httpclient>)

set(_backends "")
if(HTTPCLIENT_CURL)
  find_package(CURL REQUIRED)
  find_package(Threads REQUIRED)
  target_sources(httpclient PRIVATE src/CurlClient.cpp)
  target_link_libraries(httpclient PRIVATE CURL::libcurl Threads::Threads)
  list(APPEND _backends "libcurl")
endif()
if(HTTPCLIENT_WINHTTP)
  target_sources(httpclient PRIVATE src/WinHTTPClient.cpp)
  target_link_libraries(httpclient PRIVATE winhttp)
  list(APPEND _backends "WinHTTP")
endif()
if(HTTPCLIENT_ANDROID)
  target_sources(httpclient PRIVATE src/AndroidClient.cpp)
  list(APPEND _backends "Android")
endif()
if(HTTPCLIENT_CFNETWORK)
  target_sources(httpclient PRIVATE src/iOSCFClient.cpp)
  target_link_libraries(httpclient PRIVATE "-framework CoreFoundation" "-framework CFNetwork")
  list(APPEND _backends "CFNetwork")
endif()
if(HTTPCLIENT_NSURL)
  enable_language(OBJCXX)
  target_sources(httpclient PRIVATE src/iOSClient.mm)
  target_link_libraries(httpclient PRIVATE "-framework Foundation")
  list(APPEND _backends "NSURLRequest")
endif()
if(NOT _backends)
  message(WARNING "httpclient: no backend is enabled, so there is no HTTPClientFactory to create a client with")
endif()
string(REPLACE ";" ", " _backends_text "${_backends}")
message(STATUS "httpclient backends: ${_backends_text}")

if(HTTPCLIENT_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

if(HTTPCLIENT_INSTALL AND NOT _utf8proc_system)
  message(STATUS "httpclient: no install rules, because utf8proc was fetched, not found on the system")
  set(HTTPCLIENT_INSTALL OFF)
endif()
if(HTTPCLIENT_INSTALL)
  include(GNUInstallDirs)
  include(CMakePackageConfigHelpers)
  install(TARGETS httpclient EXPORT httpclientTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/httpclient FILES_MATCHING PATTERN "*.h")
  install(EXPORT httpclientTargets NAMESPACE httpclient:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/httpclient)
  # The package file looks for what the installed library was built against.
  set(HTTPCLIENT_FIND_CURL ${HTTPCLIENT_CURL})
  set(HTTPCLIENT_FIND_THREADS ${HTTPCLIENT_CURL})
  configure_package_config_file(cmake/httpclientConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/httpclientConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/httpclient)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/httpclientConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/httpclient)
endif()
